public class Main {
  public static void main(String[] a) {
    double[] d = {12.5,7.0,23.75,4.5,18.0,9.25,31.0};
    double sum=0, min=d[0], max=d[0];
    for (double v : d) { sum+=v; min=Math.min(min,v); max=Math.max(max,v); }
    System.out.printf("count : %d%n", d.length);
    System.out.printf("sum   : %.2f%n", sum);
    System.out.printf("mean  : %.2f%n", sum/d.length);
    System.out.printf("min   : %.2f%n", min);
    System.out.printf("max   : %.2f%n", max);
  }
}
